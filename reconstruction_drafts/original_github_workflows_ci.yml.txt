name: ci

on:
  push:
  pull_request:

jobs:
  check:
    runs-on: ubuntu-latest
    env:
      PYTHONUTF8: "1"
    steps:
      - uses: actions/checkout@v4
      - uses: astral-sh/setup-uv@v6
      - run: uv sync --all-extras --locked
      - run: uv run ruff check src examples
      - run: uv run ruff format --check src examples
      - run: uv run pyright
      - name: unit tests
        run: |
          if [ -d tests/unit ]; then uv run pytest tests/unit -m "slow or not slow"; else echo "no tests yet"; fi
      - name: regression check against the committed baseline
        run: |
          uv run aea run examples/calculator/dataset.jsonl --agent examples.calculator.agent:build_agent --runs-dir runs
          uv run aea compare baselines/calculator.json runs/*/ --output comparison.md
          uv run aea run examples/calculator/dataset.jsonl --adapter langgraph --agent examples.calculator.langgraph_agent:build_agent --runs-dir runs-langgraph
          uv run aea compare baselines/calculator.json runs-langgraph/*/ --output comparison-langgraph.md
          uv run aea run examples/orders/dataset.jsonl --agent examples.orders.agent:build_agent --runs-dir runs-orders
          uv run aea compare baselines/orders.json runs-orders/*/ --output comparison-orders.md
          uv run aea run examples/expense/dataset.jsonl --adapter langgraph --agent examples.expense.agent:build_agent --scorer examples.expense.scorers:amounts_grounded --runs-dir runs-expense
          uv run aea compare baselines/expense runs-expense/*/ --output comparison-expense.md
      - if: always()
        run: cat comparison*.md >> "$GITHUB_STEP_SUMMARY" 2>/dev/null || true
